cmake_minimum_required(VERSION 3.24)
project(XmlMergeTool VERSION 3.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(XMLMERGETOOL_BUILD_APP "Build the Qt desktop application" ON)
include(CTest)
if(MSVC)
  add_compile_options(/utf-8 /permissive-)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(LibXml2 REQUIRED)
find_package(ZLIB REQUIRED)
set(RESOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../resources/robot/htmldata")
file(GLOB_RECURSE REPORT_ASSETS CONFIGURE_DEPENDS "${RESOURCE_ROOT}/*.js" "${RESOURCE_ROOT}/*.css" "${RESOURCE_ROOT}/*.html")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${REPORT_ASSETS})
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(EMBEDDED "${CMAKE_CURRENT_BINARY_DIR}/generated/resources.hpp")
file(WRITE "${EMBEDDED}" "// Generated by CMake. Do not edit.\n#pragma once\n#include <map>\n#include <string>\n#include <string_view>\n#include <vector>\ninline const std::map<std::string, std::vector<std::string_view>> report_resources = {\n")
foreach(ASSET IN LISTS REPORT_ASSETS)
  file(RELATIVE_PATH KEY "${RESOURCE_ROOT}" "${ASSET}")
  file(READ "${ASSET}" CONTENT HEX)
  file(APPEND "${EMBEDDED}" "{\"${KEY}\", {\n")
  string(LENGTH "${CONTENT}" LENGTH)
  set(OFFSET 0)
  while(OFFSET LESS LENGTH)
    string(SUBSTRING "${CONTENT}" ${OFFSET} 8192 CHUNK)
    string(REGEX REPLACE "(..)" "\\\\x\\1" CHUNK "${CHUNK}")
    file(APPEND "${EMBEDDED}" "\"${CHUNK}\",\n")
    math(EXPR OFFSET "${OFFSET} + 8192")
  endwhile()
  file(APPEND "${EMBEDDED}" "}},\n")
endforeach()
file(APPEND "${EMBEDDED}" "};\n")
add_library(rf_report_core src/format.cpp src/reader.cpp src/encoder.cpp src/html.cpp)
target_include_directories(rf_report_core PUBLIC src PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(rf_report_core PUBLIC LibXml2::LibXml2 ZLIB::ZLIB)
if(MSVC)
  target_compile_options(rf_report_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(rf_report_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(XMLMERGETOOL_BUILD_APP)
  include(cmake/Dependencies.cmake)
  find_package(Threads REQUIRED)
  find_package(Qt6 6.9 REQUIRED COMPONENTS Widgets Concurrent)
  add_library(xml_merge_core app/merge_service.cpp)
  target_include_directories(xml_merge_core PUBLIC app)
  target_link_libraries(xml_merge_core PUBLIC rf_report_core PRIVATE xlsxwriter miniz Threads::Threads)

  add_library(xml_merge_ui STATIC
    ui/main_window.cpp ui/main_window.hpp
    ui/candidate_view.cpp ui/candidate_view.hpp ui/ui.qrc)
  set_target_properties(xml_merge_ui PROPERTIES AUTOMOC ON AUTORCC ON)
  target_include_directories(xml_merge_ui PUBLIC ui)
  target_link_libraries(xml_merge_ui PUBLIC xml_merge_core Qt6::Widgets Qt6::Concurrent)
  if(APPLE)
    enable_language(OBJCXX)
    target_sources(xml_merge_ui PRIVATE ui/macos_accessibility.mm)
    target_link_libraries(xml_merge_ui PRIVATE "-framework AppKit")
  endif()

  qt_add_executable(XmlMergeTool WIN32 MACOSX_BUNDLE "${CMAKE_CURRENT_SOURCE_DIR}/ui/main.cpp")
  target_link_libraries(XmlMergeTool PRIVATE xml_merge_ui)
  set_target_properties(XmlMergeTool PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER us.cceniam.xmlmergetool
    MACOSX_BUNDLE_BUNDLE_NAME XmlMergeTool
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
  if(WIN32)
    set_target_properties(XmlMergeTool PROPERTIES OUTPUT_NAME XmlMergeTool-app)
    configure_file(cmake/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    target_sources(XmlMergeTool PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc"
      "${CMAKE_CURRENT_SOURCE_DIR}/../scripts/windows/app.manifest")
  endif()
  if(APPLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tmp/XmlMergeTool.icns")
    target_sources(XmlMergeTool PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../tmp/XmlMergeTool.icns")
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/../tmp/XmlMergeTool.icns"
      PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    set_target_properties(XmlMergeTool PROPERTIES MACOSX_BUNDLE_ICON_FILE XmlMergeTool.icns)
  endif()
  install(TARGETS XmlMergeTool BUNDLE DESTINATION . COMPONENT Runtime
    RUNTIME DESTINATION bin COMPONENT Runtime)
  if(APPLE)
    # macdeployqt scans every plugin in the SDK by default, including plugins
    # from unrelated Homebrew Qt modules. Select the Widgets runtime explicitly.
    set(deploy_modules "")
    foreach(plugin_spec "QCocoaIntegrationPlugin|platforms" "QICOPlugin|imageformats" "QMacStylePlugin|styles")
      string(REPLACE "|" ";" plugin_parts "${plugin_spec}")
      list(GET plugin_parts 0 plugin_target)
      list(GET plugin_parts 1 plugin_directory)
      set(plugin_destination "XmlMergeTool.app/Contents/PlugIns/${plugin_directory}")
      # Homebrew's aggregate Qt SDK exposes plugins as relative symlinks into
      # its Cellar. Copy their real files so the installed bundle is standalone.
      install(CODE "
file(REAL_PATH \"$<TARGET_FILE:Qt6::${plugin_target}>\" plugin_source)
file(INSTALL DESTINATION \"\${CMAKE_INSTALL_PREFIX}/${plugin_destination}\"
  TYPE FILE FILES \"\${plugin_source}\"
  RENAME \"$<TARGET_FILE_NAME:Qt6::${plugin_target}>\")
" COMPONENT Runtime)
      string(APPEND deploy_modules "\n    \"${plugin_destination}/$<TARGET_FILE_NAME:Qt6::${plugin_target}>\"")
    endforeach()
    qt_generate_deploy_script(TARGET XmlMergeTool OUTPUT_SCRIPT app_deploy CONTENT "
qt_deploy_runtime_dependencies(
  EXECUTABLE \"XmlMergeTool.app\"
  NO_PLUGINS
  ADDITIONAL_MODULES ${deploy_modules})
")
  else()
    qt_generate_deploy_app_script(TARGET XmlMergeTool OUTPUT_SCRIPT app_deploy
      NO_UNSUPPORTED_PLATFORM_ERROR)
  endif()
  install(SCRIPT ${app_deploy} COMPONENT Runtime)
  if(WIN32)
    # vcpkg's app-local step resolves the non-Qt DLL closure beside our .exe.
    # windeployqt does not deploy those libraries, so copy them explicitly.
    install(DIRECTORY "$<TARGET_FILE_DIR:XmlMergeTool>/" DESTINATION bin
      COMPONENT Runtime FILES_MATCHING PATTERN "*.dll")
    # Qt may deploy vc_redist.exe, but a portable application must not need
    # an administrator to install it. Deploy the licensed app-local CRT.
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Runtime)
    include(InstallRequiredSystemLibraries)
  endif()
  if(APPLE)
    set(notice_destination XmlMergeTool.app/Contents/Resources/licenses)
  else()
    set(notice_destination bin/licenses)
  endif()
  install(FILES THIRD_PARTY_NOTICES.md ROBOT_FRAMEWORK_LICENSE.txt
    "${miniz_SOURCE_DIR}/LICENSE" DESTINATION ${notice_destination} COMPONENT Runtime)
  install(FILES "${xlsxwriter_SOURCE_DIR}/License.txt"
    DESTINATION ${notice_destination} RENAME libxlsxwriter.txt COMPONENT Runtime)
  install(DIRECTORY licenses/ DESTINATION ${notice_destination} COMPONENT Runtime)
  if(WIN32 AND VCPKG_INSTALLED_DIR)
    foreach(dependency libxml2 zlib libiconv)
      set(copyright_file "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${dependency}/copyright")
      if(EXISTS "${copyright_file}")
        install(FILES "${copyright_file}" DESTINATION ${notice_destination}
          RENAME "${dependency}-copyright.txt" COMPONENT Runtime)
      endif()
    endforeach()
  endif()

  if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(xml-merge-driver tests/app_driver.cpp)
    target_link_libraries(xml-merge-driver PRIVATE xml_merge_core)
    if(WIN32)
      target_link_libraries(xml-merge-driver PRIVATE shell32)
    endif()
    add_executable(xml-merge-ui-tests tests/ui_tests.cpp)
    set_target_properties(xml-merge-ui-tests PROPERTIES AUTOMOC ON)
    target_link_libraries(xml-merge-ui-tests PRIVATE xml_merge_ui Qt6::Test)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/Testing/Temporary")
    add_test(NAME xml-merge-ui-tests COMMAND xml-merge-ui-tests
      -o -,txt -o "${CMAKE_CURRENT_BINARY_DIR}/Testing/Temporary/xml-merge-ui-tests.txt,txt")
    # QtTest uses OutputDebugString on Windows when CTest launches it without a
    # controlling console. Preserve both captured output and an artifact log.
    set_tests_properties(xml-merge-ui-tests PROPERTIES
      ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_FORCE_STDERR_LOGGING=1"
      TIMEOUT 120)
    if(WIN32)
      # CTest runs before deployment, so resolve the SDK's Qt DLLs for this
      # test without requiring callers to change their PATH.
      set_tests_properties(xml-merge-ui-tests PROPERTIES
        ENVIRONMENT_MODIFICATION
          "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    endif()
  endif()
endif()
add_executable(rf-report-cpp src/main.cpp)
target_link_libraries(rf-report-cpp PRIVATE rf_report_core)
if(WIN32)
  target_link_libraries(rf-report-cpp PRIVATE shell32)
endif()
install(TARGETS rf-report-cpp RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_executable(rf-report-unit tests/unit.cpp)
  target_link_libraries(rf-report-unit PRIVATE rf_report_core)
  add_test(NAME rf-report-unit COMMAND rf-report-unit)
  set_tests_properties(rf-report-unit PROPERTIES TIMEOUT 30)
endif()
